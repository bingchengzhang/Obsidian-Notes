void PercolateUp( int p, PriorityQueue H )
{
	int temp = H->Elements[p];
	int i;
	for (i = p; temp < H->Elements[i/2]; i /= 2 )//parent > child -->percolateup

		H->Elements[i] = H->Elements[i/2];

void PercolateDown( int p, PriorityQueue H )
{
	int child;
	int temp = H->Elements[p];
	for ( ; p*2 <= H->Size; p = child )
	{
		child = 2*p;//default child is left
		if ( child+1 <= H->Size )//have right
		{	
			if ( H->Elements[child] > H->Elements[child+1] )
			child++;//right child is smaller
		}
		if ( temp > H->Elements[child] )
		{
			H->Elements[p] = H->Elements[child];
		}
		else
		break;//p = child/2
	}
	H->Elements[p] = temp;
}
